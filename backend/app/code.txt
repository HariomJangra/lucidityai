Realtime tool-aware streaming (summary + code to apply)

Goal
- Frontend shows tool progress only when the agent actually calls a tool.
- "Searching the web" is replaced by the actual tool name (web_search, calculator, etc.).
- Simple prompts (ex: "Hello") can answer directly without any tool steps.
- Streaming uses SSE at /search/stream.

```
from fastapi.responses import StreamingResponse

from app.services.agent import run_agent, run_agent_stream

router = APIRouter()


@router.get("/search")
def search(q: str = Query(..., min_length=1)):
		response = run_agent(q)
		return {"response": response}


@router.get("/search/stream")
def search_stream(q: str = Query(..., min_length=1)):
		def event_generator():
				for event in run_agent_stream(q):
						payload = json.dumps(event["data"], ensure_ascii=True)
						yield f"event: {event['event']}\n"
						yield f"data: {payload}\n\n"

		return StreamingResponse(
				event_generator(),
				media_type="text/event-stream",
				headers={
						"Cache-Control": "no-cache",
						"Connection": "keep-alive",
				},
		)
```

```
import hashlib
import json
from concurrent.futures import ThreadPoolExecutor, as_completed
from typing import Any, Dict, List, Tuple

from app.core.config import get_settings

import numpy as np
import requests
import trafilatura
from huggingface_hub import InferenceClient

settings = get_settings()

# Configuration & API Setup
client = InferenceClient(api_key=settings.huggingface_api_key)

BI_ENCODER_MODEL = "BAAI/bge-base-en-v1.5"
CROSS_ENCODER_MODEL = "BAAI/bge-reranker-base"


# Text Processing & Chunking
def chunk_text(text, chunk_size=400, overlap=50):
		words = text.split()
		chunks = []
		step = chunk_size - overlap

		for i in range(0, len(words), step):
				chunk = " ".join(words[i : i + chunk_size])
				if chunk:
						chunks.append(chunk)

		return chunks


def build_chunks(content: dict) -> list:
		all_chunks = []

		for url, text in content.items():
				if not text or text.startswith("Error:"):
						continue

				domain = url.split("//")[-1].split("/")[0].replace("www.", "")
				raw_chunks = chunk_text(text)

				for idx, chunk_text_str in enumerate(raw_chunks):
						chunk_id = hashlib.md5(f"{url}-{idx}".encode()).hexdigest()[:12]

						all_chunks.append(
								{
										"chunk_id": chunk_id,
										"url": url,
										"domain": domain,
										"text": chunk_text_str,
										"chunk_index": idx,
										"total_chunks": len(raw_chunks),
										"metadata": {
												"char_count": len(chunk_text_str),
												"word_count": len(chunk_text_str.split()),
										},
								}
						)

		return all_chunks


# Hugging Face API Embeddings & Reranking

def bi_encode_chunks(chunks: list, batch_size=32) -> np.ndarray:
		"""Fetches embeddings from HF Serverless API in manageable batches."""
		texts = [c["text"] for c in chunks]
		all_embeddings = []

		# Batching prevents hitting payload size limitations
		for i in range(0, len(texts), batch_size):
				batch = texts[i : i + batch_size]
				try:
						embeddings = client.feature_extraction(
								text=batch, model=BI_ENCODER_MODEL
						)
						all_embeddings.extend(embeddings)
				except Exception as e:
						print(f"Error extracting features for batch {i}: {e}")
						# Fallback to zero vectors if a batch completely fails
						all_embeddings.extend([np.zeros(768) for _ in batch])

		embedding_matrix = np.array(all_embeddings)

		# Normalize embeddings manually
		norms = np.linalg.norm(embedding_matrix, axis=1, keepdims=True)
		# Avoid division by zero
		norms[norms == 0] = 1.0

		return embedding_matrix / norms


def retrieve_top_k(
		query: str, chunks: list, chunk_embeddings: np.ndarray, k=50
):
		"""Fetches query embedding from HF and performs cosine similarity locally."""
		try:
				query_emb = np.array(
						client.feature_extraction(text=query, model=BI_ENCODER_MODEL)
				)
		except Exception as e:
				print(f"Error fetching query embedding: {e}")
				return []

		# Compute dot product similarity matrix (since vectors are normalized)
		scores = np.dot(chunk_embeddings, query_emb.T).squeeze()

		# Handle edge case where only 1 chunk exists
		if scores.ndim == 0:
				scores = np.array([scores])

		top_k_idx = np.argsort(scores)[::-1][:k]

		return [(chunks[i], float(scores[i])) for i in top_k_idx]


def rerank(query: str, candidates: list, top_n=5):
		"""Reranks candidates via HF Text Classification API (CrossEncoder)."""
		ranked_results = []

		for chunk, bi_score in candidates:
				try:
						# Cross-encoders act as Text Classification tasks on Hugging Face Serverless API
						api_response = client.post(
								json={"inputs": {"text": query, "text_pair": chunk["text"]}},
								model=CROSS_ENCODER_MODEL,
						)

						response_data = json.loads(api_response.decode("utf-8"))

						# Serverless API maps outputs to a standard label probability array
						# The top class score reflects the similarity/relevance score
						ce_score = response_data["score"]

				except Exception as e:
						# Fallback to 0 if API rate limits or drops a payload chunk
						ce_score = 0.0

				ranked_results.append((chunk, bi_score, float(ce_score)))

		# Sort candidates dynamically by their ce_score
		ranked = sorted(ranked_results, key=lambda x: x[2], reverse=True)

		return [
				{**chunk, "bi_score": bi_score, "ce_score": ce_score}
				for chunk, bi_score, ce_score in ranked[:top_n]
		]


def search(
		query: str, chunks: list, chunk_embeddings: np.ndarray, k=50, top_n=5
):
		candidates = retrieve_top_k(query, chunks, chunk_embeddings, k=k)

		if not candidates:
				return []

		results = rerank(query, candidates, top_n=top_n)
		return results


# Web Scraping & Main Search Control Flow
def scrape_url(url):
		try:
				downloaded = trafilatura.fetch_url(url)
				result = trafilatura.extract(
						downloaded, include_comments=False, include_tables=False
				)
				return url, result
		except Exception as e:
				return url, f"Error: {e}"


def fetch_search_results(query: str) -> List[Dict[str, Any]]:
		response = requests.get(
				"http://searxng-railway-production-5eae.up.railway.app/search",
				params={
						"q": query,
						"format": "json",
						"categories": "general",
						"language": "en-IN",
						"time_range": "month",
						"safesearch": 0,
						"pageno": 1,
				},
				timeout=20,
		)

		data = response.json()
		return data.get("results", [])


def extract_links(
		results: List[Dict[str, Any]], limit: int = 7
) -> Tuple[List[Dict[str, Any]], List[str], Dict[str, Dict[str, Any]]]:
		links: List[Dict[str, Any]] = []
		urls: List[str] = []
		meta_by_url: Dict[str, Dict[str, Any]] = {}

		for item in results:
				url = item.get("url")
				if not url or url in meta_by_url:
						continue

				domain = url.split("//")[-1].split("/")[0].replace("www.", "")
				title = item.get("title") or item.get("content") or domain
				snippet = item.get("content") or item.get("snippet")

				link = {
						"title": title,
						"url": url,
						"domain": domain,
						"logo": "globe",
						"snippet": snippet,
				}

				links.append(link)
				urls.append(url)
				meta_by_url[url] = link

				if len(urls) >= limit:
						break

		return links, urls, meta_by_url


def build_context(
		query: str, urls: List[str], top_n: int = 5
) -> Tuple[str, List[str]]:
		if not urls:
				return "", []

		content: Dict[str, str] = {}

		with ThreadPoolExecutor(max_workers=10) as executor:
				futures = [executor.submit(scrape_url, url) for url in urls]
				for future in as_completed(futures):
						url, result = future.result()
						content[url] = result

		chunks = build_chunks(content)

		if not chunks:
				return "", []

		chunk_embeddings = bi_encode_chunks(chunks)
		candidates = retrieve_top_k(query, chunks, chunk_embeddings, k=50)

		results = rerank(query, candidates, top_n=top_n) if candidates else []
		if not results:
				return "", []

		url_to_idx: Dict[str, int] = {}
		ordered_urls: List[str] = []
		context_lines: List[str] = []

		for r in results:
				url = r["url"]
				if url not in url_to_idx:
						url_to_idx[url] = len(url_to_idx) + 1
						ordered_urls.append(url)

				idx = url_to_idx[url]
				context_lines.append(f"[{idx}] {url}\n{r['text']}")

		context = "\n\n".join(context_lines)
		return context, ordered_urls


def websearch_structured(query: str) -> Dict[str, Any]:
		results = fetch_search_results(query)
		links, urls, meta_by_url = extract_links(results)

		if not urls:
				return {
						"context": "",
						"links": [],
						"citations": [],
				}

		context, ordered_urls = build_context(query, urls, top_n=5)

		citations: List[Dict[str, Any]] = []
		for idx, url in enumerate(ordered_urls, start=1):
				meta = meta_by_url.get(url) or {}
				domain = meta.get("domain") or url.split("//")[-1].split("/")[0].replace("www.", "")
				citations.append(
						{
								"id": idx,
								"domain": domain,
								"url": url,
								"title": meta.get("title"),
								"snippet": meta.get("snippet"),
						}
				)

		return {
				"context": context,
				"links": links,
				"citations": citations,
		}


def websearch(query: str) -> str:

		structured = websearch_structured(query)
		if not structured["context"]:
				return "No clean text content could be processed from the search results."

		return structured["context"]
```

4) backend/app/services/agent.py (tool-aware streaming)
```
import queue
import threading
from typing import Any, Dict, Generator, List

from app.core.config import get_settings
from app.core.prompts import SYSTEM_PROMPT
from langchain.chat_models import init_chat_model
from langchain.agents import create_agent
from langchain.messages import HumanMessage, SystemMessage
from langchain.tools import tool
from app.services.tools import web_search, calculator, fetch_url, code_execution, chart_visualization
from app.services.websearch import websearch_structured

settings = get_settings()

model = init_chat_model("mistral-small-latest", model_provider="mistralai", api_key=settings.mistral_api_key)

agent = create_agent(model=model, tools=[web_search, calculator, fetch_url, code_execution, chart_visualization])


def run_agent(user_message: str) -> str:
	result: Any = agent.invoke(
		{
			"messages": [
				SystemMessage(content=SYSTEM_PROMPT),
				HumanMessage(content=user_message),
			]
		}
	)

	if isinstance(result, dict):
		if "output" in result:
			return str(result["output"])
		messages = result.get("messages")
		if messages:
			last_message = messages[-1]
			return str(getattr(last_message, "content", last_message))

	return str(getattr(result, "content", result))


def run_agent_stream(user_message: str) -> Generator[Dict[str, Any], None, None]:
	def emit(event: str, data: Dict[str, Any]) -> Dict[str, Any]:
		return {"event": event, "data": data}

	event_queue: "queue.Queue[Dict[str, Any]]" = queue.Queue()
	done_event = threading.Event()

	tool_links: List[Dict[str, Any]] = []
	search_links: List[Dict[str, Any]] = []
	citations: List[Dict[str, Any]] = []
	step_title = ""

	def push(event: str, data: Dict[str, Any]) -> None:
		event_queue.put(emit(event, data))

	def record_tool(name: str, input_value: Any, add_entry: bool = True) -> None:
		nonlocal step_title
		step_title = name
		push("tool", {"name": name, "input": str(input_value) if input_value is not None else ""})
		push("status", {"status": "searching"})
		if add_entry:
			tool_links.append(
				{
					"title": name,
					"url": f"tool:{name}",
					"domain": "tool",
					"logo": "globe",
					"snippet": str(input_value) if input_value is not None else "",
				}
			)

	@tool(
		"web_search",
		description="A tool to perform a web search and retrieve relevant information. Input should be a search query as a string, e.g. \"What is the capital of France?\".",
	)
	def web_search_stream(query: str) -> str:
		record_tool("web_search", query, add_entry=False)
		structured = websearch_structured(query)
		links = structured.get("links", [])
		search_links.extend(links)
		citations.extend(structured.get("citations", []))

		for link in links:
			push("link", {"link": link})

		if structured.get("context"):
			push("status", {"status": "analyzing"})
			return structured["context"]

		return "No search results found."

	@tool(
		"calculator",
		description="A tool to evaluate mathematical expressions. Input should be a valid math expression as a string, e.g. \"2 + 2 * (3 - 1)\".",
	)
	def calculator_stream(expression: str) -> str:
		record_tool("calculator", expression)
		result = calculator.invoke(expression)
		push("status", {"status": "thinking"})
		return result

	@tool(
		"fetch_url",
		description="A tool to fetch and extract text content from a given URL. Input should be a valid URL as a string, e.g. \"https://www.example.com\".",
	)
	def fetch_url_stream(url: str) -> str:
		record_tool("fetch_url", url)
		result = fetch_url.invoke(url)
		push("status", {"status": "thinking"})
		return result

	@tool(
		"code_execution",
		description="A tool to execute code snippets. Input should be a valid code as a string.",
	)
	def code_execution_stream(code: str, language: str = "python", input: str = "") -> str:
		record_tool("code_execution", f"{language} code")
		result = code_execution.invoke({"code": code, "language": language, "input": input})
		push("status", {"status": "thinking"})
		return result

	@tool(
		"chart_visualization",
		description="A tool to create charts from data. Input should be a JSON string containing the data and chart type, e.g. {\"type\": \"bar\", \"data\": {\"labels\": [\"A\", \"B\"], \"values\": [10, 20]}}.",
	)
	def chart_visualization_stream(data: str) -> str:
		record_tool("chart_visualization", "chart data")
		result = chart_visualization.invoke(data)
		push("status", {"status": "thinking"})
		return result if result is not None else ""

	stream_agent = create_agent(
		model=model,
		tools=[
			web_search_stream,
			calculator_stream,
			fetch_url_stream,
			code_execution_stream,
			chart_visualization_stream,
		],
	)

	def run_agent_task() -> None:
		try:
			push("status", {"status": "thinking"})
			result = stream_agent.invoke(
				{
					"messages": [
						SystemMessage(content=SYSTEM_PROMPT),
						HumanMessage(content=user_message),
					]
				}
			)

			if isinstance(result, dict):
				if "output" in result:
					answer = str(result["output"])
				else:
					messages = result.get("messages")
					last_message = messages[-1] if messages else result
					answer = str(getattr(last_message, "content", last_message))
			else:
				answer = str(getattr(result, "content", result))

			links = search_links if search_links else tool_links
			steps: List[Dict[str, Any]] = []
			if links:
				steps = [
					{
						"title": step_title or "tools",
						"query": user_message,
						"links": links,
					}
				]

			result_data = {
				"title": user_message,
				"tab": "Answer",
				"steps": steps,
				"answer": answer,
				"citations": citations,
				"related": [],
				"images": [],
			}

			push("status", {"status": "typing"})
			chunk_size = 12
			for i in range(0, len(answer), chunk_size):
				push("token", {"text": answer[i : i + chunk_size]})

			push("done", {"result": result_data})
		except Exception as exc:
			push("error", {"message": str(exc)})
		finally:
			done_event.set()

	threading.Thread(target=run_agent_task, daemon=True).start()

	while not done_event.is_set() or not event_queue.empty():
		try:
			event = event_queue.get(timeout=0.1)
			yield event
		except queue.Empty:
			continue
```

```
	switch (name) {
		case 'web_search':
			return 'Web Search'
		case 'calculator':
			return 'Calculator'
		case 'fetch_url':

		case 'chart_visualization':

const eventSource = new EventSource(
	`${API_BASE}/search/stream?q=${encodeURIComponent(queryText)}`
)

eventSource.addEventListener('status', (event) => {
	const data = JSON.parse((event as MessageEvent).data)
	const status = data.status as ChatTurn['status']
	updateTurn((t) => ({
		...t,
		status,
		stepsExpanded: status === 'searching' ? true : status === 'analyzing' ? false : t.stepsExpanded
	}))
})

eventSource.addEventListener('tool', (event) => {
	const data = JSON.parse((event as MessageEvent).data)
	const toolName = data.name as string
	const toolInput = data.input as string
	const toolLabel = formatToolLabel(toolName)

	updateTurn((t) => {
		if (!t.resultData) return t

		const steps = t.resultData.steps.length
			? [...t.resultData.steps]
			: [{ title: toolLabel, query: t.query, links: [] }]

		const stepZero = steps[0]
		const isWebSearch = toolName === 'web_search'
		const toolEntry = {
			title: toolLabel,
			url: `tool:${toolName}`,
			domain: 'tool',
			logo: 'globe',
			snippet: toolInput
		}

		const updatedLinks = isWebSearch ? stepZero.links : [...stepZero.links, toolEntry]

		steps[0] = {
			...stepZero,
			title: toolLabel,
			links: updatedLinks
		}

		return {
			...t,
			resultData: { ...t.resultData, steps },
			stepsExpanded: true,
			visibleLinksCount: updatedLinks.length
		}
	})
})

eventSource.addEventListener('link', (event) => {
	const data = JSON.parse((event as MessageEvent).data)
	const link = data.link

	updateTurn((t) => {
		if (!t.resultData) return t
		const steps = [...t.resultData.steps]
		const stepZero = steps[0] || { title: 'Searching the web', query: t.query, links: [] }
		const updatedLinks = [...stepZero.links, link]
		steps[0] = { ...stepZero, links: updatedLinks }

		return {
			...t,
			resultData: { ...t.resultData, steps },
			visibleLinksCount: updatedLinks.length
		}
	})
})

eventSource.addEventListener('token', (event) => {
	const data = JSON.parse((event as MessageEvent).data)
	const text = data.text || ''

	updateTurn((t) => ({
		...t,
		status: 'typing',
		streamedAnswer: `${t.streamedAnswer}${text}`
	}))
})

eventSource.addEventListener('fault', (event) => {
	const data = JSON.parse((event as MessageEvent).data)
	const message = data.message || 'An unexpected error occurred.'

	updateTurn((t) => {
		const notice = `Notice: ${message}`
		const nextText = t.streamedAnswer
			? `${t.streamedAnswer}\n\n${notice}`
			: notice

		return {
			...t,
			streamedAnswer: nextText,
			status: t.status === 'searching' ? 'typing' : t.status
		}
	})
})

eventSource.addEventListener('done', (event) => {
	const data = JSON.parse((event as MessageEvent).data)
	const result = data.result as ResultData

	updateTurn((t) => ({
		...t,
		status: 'completed',
		resultData: result,
		streamedAnswer: result.answer || t.streamedAnswer,
		visibleLinksCount: result.steps?.[0]?.links?.length || t.visibleLinksCount,
		stepsExpanded: false
	}))

	eventSource.close()
})

eventSource.onerror = () => {
	updateTurn((t) => ({
		...t,
		status: 'completed',
		streamedAnswer: t.streamedAnswer || 'Streaming failed. Please try again.'
	}))
	eventSource.close()
}
```

```
									className="tree-item link-item animate-slide-up-gently"
									style={{ animationDuration: '0.3s' }}
								>
									<span className="tree-item-icon">
										{renderSourceIcon(link.logo)}
									</span>
									<span className="tree-item-title">{link.title}</span>
									<span className="tree-item-domain">tool</span>
								</div>
							)
						}

						return (
							<a 
								key={lIdx} 
								href={link.url} 
								target="_blank" 
								rel="noopener noreferrer" 
								className="tree-item link-item animate-slide-up-gently"
								style={{ animationDuration: '0.3s' }}
								onMouseEnter={(e) => handleTriggerMouseEnter(e, turn, link.domain)}
								onMouseLeave={handleTriggerMouseLeave}
							>
								<span className="tree-item-icon">
									{renderSourceIcon(link.logo)}
								</span>
								<span className="tree-item-title">{link.title}</span>
								<span className="tree-item-domain">{link.domain}</span>
							</a>
						)
					})}

					{turn.resultData.steps[0] && turn.visibleLinksCount >= turn.resultData.steps[0].links.length && (
						<div className="tree-item more-item animate-slide-up-gently" style={{ animationDuration: '0.3s' }} onClick={() => setActiveResultTab('Links')}>
							<span className="more-text">+7 more</span>
						</div>
					)}
				</div>
			</div>
		)}
	</div>
)}
```
